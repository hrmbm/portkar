// UPDATE: Proses SEMUA kandidat yang belum pernah diproses (Status PDF != 'Sudah')
// ============================================================================
function processNewCandidate() {
  var sheet = getResponseSheet_();
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) {
    notify_('Belum ada jawaban responden.');
    return;
  }

  var lastColumn = sheet.getLastColumn();
  var headers = sheet.getRange(1, 1, 1, lastColumn).getValues()[0];
  
  // Pastikan kolom "Status PDF" ada
  var statusColIdx = headers.indexOf('Status PDF');
  if (statusColIdx === -1) {
    statusColIdx = headers.length;
    sheet.getRange(1, statusColIdx + 1).setValue('Status PDF');
    headers.push('Status PDF');
  }

  // Ambil semua nilai di kolom Status PDF untuk dicek
  var statusValues = sheet.getRange(2, statusColIdx + 1, lastRow - 1, 1).getValues();
  
  var processedCount = 0;
  var errorCount = 0;
  var lastProcessedName = '';

  for (var r = 0; r < statusValues.length; r++) {
    var row = r + 2; // Baris data dimulai dari baris ke-2
    var statusValue = String(statusValues[r][0] || '').trim().toLowerCase();
    
    // Hanya proses jika statusnya belum 'sudah'
    if (statusValue !== 'sudah') {
      try {
        var result = processRow_(sheet, row);
        if (result) {
          processedCount++;
          // Ambil nama untuk laporan notifikasi
          var values = sheet.getRange(row, 1, 1, lastColumn).getValues()[0];
          var nama = cariKolomAny_(headers, values, ['nama lengkap', 'nama'], null);
          lastProcessedName = nama ? String(nama).trim() : 'Tanpa Nama';
        }
      } catch (e) {
        errorCount++;
        Logger.log('Error pada baris ' + row + ': ' + e.toString());
      }
    }
  }

  // Notifikasi hasil akhir batch processing
  if (processedCount === 0 && errorCount === 0) {
    notify_('Semua kandidat sudah diproses. Tidak ada kandidat baru.');
  } else {
    var msg = 'Proses batch selesai.\nBerhasil diproses: ' + processedCount + ' kandidat.';
    if (errorCount > 0) {
      msg += '\nGagal: ' + errorCount + ' kandidat (periksa Log eksekusi).';
    }
    if (processedCount > 0) {
      msg += '\nTerakhir diproses: ' + lastProcessedName;
    }
    notify_(msg);
  }
}

function processSelectedRow() {
  var sheet = getResponseSheet_();
  var range = sheet.getActiveRange();
  if (!range) {
    notify_('Pilih salah satu baris responden terlebih dahulu.');
    return;
  }
  var row = range.getRow();
  if (row < 2) {
    notify_('Pilih baris responden (bukan header).');
    return;
  }
  return processRow_(sheet, row);
}

function getResponseSheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  return ss.getSheetByName('Responden') || ss.getSheets()[0];
}

function notify_(msg) {
  try {
    SpreadsheetApp.getUi().alert(msg);
  } catch (e) {
    Logger.log(msg);
  }
}

// INTI: baca 1 baris jawaban -> hitung -> PDF

function processRow_(sheet, row) {
  var lastColumn = sheet.getLastColumn();
  var headers = sheet.getRange(1, 1, 1, lastColumn).getValues()[0];
  var values = sheet.getRange(row, 1, 1, lastColumn).getValues()[0];

  // ------------------------------------------------------------------------
  // Anti double PDF: gunakan kolom "Status PDF"
  // ------------------------------------------------------------------------
  var statusColIdx = headers.indexOf('Status PDF');
  if (statusColIdx === -1) {
    statusColIdx = headers.length;
    sheet.getRange(1, statusColIdx + 1).setValue('Status PDF');
    headers.push('Status PDF');
    values.push('');
  }

  var statusValue = String(values[statusColIdx] || '').trim().toLowerCase();
  if (statusValue === 'sudah') {
    notify_('Baris ini sudah pernah diproses sebelumnya.\nJika ingin diproses ulang, hapus isi kolom "Status PDF" pada baris tersebut.');
    return null;
  }

  // ------------------------------------------------------------------------
  // Ambil data diri dengan fuzzy matching
  // ------------------------------------------------------------------------
  var nama = cariKolomAny_(headers, values, ['nama lengkap', 'nama'], null);
  var usia = cariKolomAny_(headers, values, ['usia', 'umur'], null);
  var gender = cariKolomAny_(headers, values, ['kelamin', 'gender', 'jenis'], null);

  if (nama === null || String(nama).trim() === '') throw new Error('Kolom Nama tidak ditemukan atau kosong.');
  if (usia === null || String(usia).trim() === '') throw new Error('Kolom Usia tidak ditemukan atau kosong.');
  if (gender === null || String(gender).trim() === '') throw new Error('Kolom Jenis Kelamin tidak ditemukan atau kosong.');

  nama = String(nama).trim();
  usia = String(usia).trim();
  gender = String(gender).trim();

  var tglRaw = values[0] instanceof Date ? values[0] : cariKolomAny_(headers, values, ['timestamp', 'tanggal', 'tgl', 'waktu'], null);
  var tgl = tglRaw instanceof Date
    ? Utilities.formatDate(tglRaw, Session.getScriptTimeZone(), 'dd MMMM yyyy')
    : String(tglRaw || '');

  // ------------------------------------------------------------------------
  // Ambil jawaban soal 1-24 dengan fuzzy matching
  // ------------------------------------------------------------------------
  var posP = [];
  var posK = [];

  for (var q = 1; q <= 24; q++) {
    var ansP = String(ambilSoal_(headers, values, q, false) || '').trim();
    var ansK = String(ambilSoal_(headers, values, q, true) || '').trim();

    if (!ansP || !ansK) {
      throw new Error('Kolom jawaban soal ' + q + ' tidak ditemukan. Pastikan judul form mengandung nomor ' + q + ', kata PALING, dan TIDAK untuk pertanyaan kedua.');
    }

    var idxP = indexOfJawaban_(QUESTIONS[q - 1], ansP);
    var idxK = indexOfJawaban_(QUESTIONS[q - 1], ansK);

    if (idxP === -1 || idxK === -1) {
      throw new Error('Jawaban soal ' + q + ' tidak dikenali. Periksa kesesuaian teks pilihan pada Form dengan data master.\nPALING: ' + ansP + '\nTIDAK: ' + ansK);
    }

    if (idxP === idxK) {
      throw new Error('Soal ' + q + ': jawaban PALING dan PALING TIDAK sama (' + ansP + '). Data tidak valid.');
    }

    posP.push(idxP + 1);
    posK.push(idxK + 1);
  }

  // ------------------------------------------------------------------------
  // Hitung DISC
  // ------------------------------------------------------------------------
  var hasil = hitungDISC_(posP, posK);
  hasil.identitas = {
    nama: nama,
    usia: usia,
    gender: gender,
    tanggal: tgl
  };

  // Simpan ringkasan ke sheet "Hasil"
  tulisHasil_(hasil);

  // ------------------------------------------------------------------------
  // Buat PDF
  // ------------------------------------------------------------------------
  var html = buildReportHtml_(hasil);
  var safeNama = nama.replace(/[\\/:*?"<>|]/g, '');
  var pdfName = 'DISC Result - ' + safeNama + '.pdf';

  var pdfBlob = Utilities.newBlob(html, 'text/html', 'report.html')
    .getAs('application/pdf')
    .setName(pdfName);

  var folder = getOrCreateFolder_('Hasil DISC');

  // Hapus PDF lama dengan nama sama agar tidak menumpuk
  var old = folder.getFilesByName(pdfName);
  while (old.hasNext()) old.next().setTrashed(true);

  var file = folder.createFile(pdfBlob);

  // Tandai sudah diproses
  sheet.getRange(row, statusColIdx + 1).setValue('Sudah');

  notify_(
    'PDF berhasil dibuat untuk: ' + nama + '\n' +
    'Pola (MOST / LEAST / CHANGE): ' + hasil.pola[0].nama + ' / ' + hasil.pola[1].nama + ' / ' + hasil.pola[2].nama + '\n' +
    'File: ' + file.getUrl()
  );

  return file.getUrl();
}

// FUNGSI PENCARIAN HEADER FUZZY

function cariKolom_(headers, values, keywords, mustNotContain) {
  for (var i = 0; i < headers.length; i++) {
    var h = String(headers[i] || '').toLowerCase();
    if (!h) continue;

    var match = true;

    if (keywords) {
      for (var k = 0; k < keywords.length; k++) {
        if (h.indexOf(String(keywords[k]).toLowerCase()) === -1) {
          match = false;
          break;
        }
      }
    }

    if (match && mustNotContain) {
      for (var m = 0; m < mustNotContain.length; m++) {
        if (h.indexOf(String(mustNotContain[m]).toLowerCase()) !== -1) {
          match = false;
          break;
        }
      }
    }

    if (match) return values[i];
  }
  return null;
}

function cariKolomAny_(headers, values, keywords, mustNotContain) {
  for (var i = 0; i < headers.length; i++) {
    var h = String(headers[i] || '').toLowerCase();
    if (!h) continue;

    var match = false;

    if (keywords) {
      for (var k = 0; k < keywords.length; k++) {
        if (h.indexOf(String(keywords[k]).toLowerCase()) !== -1) {
          match = true;
          break;
        }
      }
    }

    if (match && mustNotContain) {
      for (var m = 0; m < mustNotContain.length; m++) {
        if (h.indexOf(String(mustNotContain[m]).toLowerCase()) !== -1) {
          match = false;
          break;
        }
      }
    }

    if (match) return values[i];
  }
  return null;
}

function ambilSoal_(headers, values, q, isTidak) {
  var numRe = new RegExp('(^|[^0-9])' + q + '\\s*\\.');

  for (var i = 0; i < headers.length; i++) {
    var h = String(headers[i] || '');
    var hl = h.toLowerCase();

    if (!numRe.test(hl)) continue;
    if (hl.indexOf('paling') === -1) continue;

    var adaTidak = hl.indexOf('tidak') !== -1;

    if (isTidak && adaTidak) return values[i];
    if (!isTidak && !adaTidak) return values[i];
  }

  return null;
}

function indexOfJawaban_(choices, answer) {
  // 🔥 FUNGSI NORMALISASI: Mengubah spasi berlebih menjadi 1 spasi, menghapus spasi di ujung, dan mengubah ke huruf kecil
  var normalize = function(s) {
    return String(s || '').replace(/\s+/g, ' ').trim().toLowerCase();
  };
  
  var a = normalize(answer);
  for (var i = 0; i < choices.length; i++) {
    if (normalize(choices[i]) === a) return i;
  }
  return -1;
}

function getOrCreateFolder_(name) {
  var target;
  try {
    target = DriveApp.getFolderById(TARGET_FOLDER_ID);
  } catch (e) {
    throw new Error('Folder target tidak ditemukan. Pastikan TARGET_FOLDER_ID benar dan akun memiliki akses ke folder tersebut.');
  }

  var it = target.getFoldersByName(name);
  return it.hasNext() ? it.next() : target.createFolder(name);
}

function hitungDISC_(posP, posK) {
  var LETTERS = ['D', 'I', 'S', 'C'];

  var cntP = { D: 0, I: 0, S: 0, C: 0 };
  var cntK = { D: 0, I: 0, S: 0, C: 0 };
  var starP = 0;
  var starK = 0;

  for (var q = 0; q < 24; q++) {
    var lp = SCORE_P[q][posP[q] - 1];
    var lk = SCORE_K[q][posK[q] - 1];

    if (lp === '*') starP++;
    else cntP[lp]++;

    if (lk === '*') starK++;
    else cntK[lk]++;
  }

  var countP = LETTERS.map(function(L) { return cntP[L]; });
  var countK = LETTERS.map(function(L) { return cntK[L]; });
  var countStarP = 24 - (countP[0] + countP[1] + countP[2] + countP[3]);
  var countStarK = 24 - (countK[0] + countK[1] + countK[2] + countK[3]);
  var diff = countP.map(function(v, i) { return v - countK[i]; });

  function vlookup(table, keys, x, col) {
    var r = table[0][col];
    for (var k = 0; k < keys.length; k++) {
      if (keys[k] <= x) r = table[k][col];
      else break;
    }
    return r;
  }

  var keysN = [];
  for (var n = 0; n <= 20; n++) keysN.push(n);

  var keysCh = [];
  for (var m = -22; m <= 22; m++) keysCh.push(m);

  var g1 = countP.map(function(v, i) { return vlookup(TABLE_MOST, keysN, v, i); });
  var g2 = countK.map(function(v, i) { return vlookup(TABLE_LEAST, keysN, v, i); });
  var g3 = diff.map(function(v, i) { return vlookup(TABLE_CHANGE, keysCh, v, i); });

  function matchPattern(g) {
    for (var p = 0; p < PATTERN_CONDITIONS.length; p++) {
      if (PATTERN_CONDITIONS[p](g[0], g[1], g[2], g[3])) return p;
    }
    return -1;
  }

  var idx = [matchPattern(g1), matchPattern(g2), matchPattern(g3)];

  var pola = idx.map(function(i) {
    if (i === -1) {
      return {
        index: -1,
        nama: 'Tidak terdefinisi',
        title: '-',
        traits: [],
        professions: '-',
        description: '-'
      };
    }
    var d = DEF_DATA[i];
    return {
      index: i + 1,
      nama: PATTERN_NAMES[i],
      title: d.title,
      traits: d.traits,
      professions: d.professions,
      description: d.description
    };
  });

  var domIdx = 0;
  for (var t = 1; t < 4; t++) {
    if (g3[t] > g3[domIdx]) domIdx = t;
  }

  var hurufDominan = LETTERS[domIdx];

  return {
    countP: countP,
    countK: countK,
    starP: countStarP,
    starK: countStarK,
    diff: diff,
    g1: g1,
    g2: g2,
    g3: g3,
    pola: pola,
    hurufDominan: hurufDominan,
    detail: DETAIL[hurufDominan]
  };
}

function esc_(s) {
  return String(s == null ? '' : s)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;');
}

function svgGraph_(values, accent) {
  var LETTERS = ['D', 'I', 'S', 'C'];
  var W = 190;
  var H = 180;
  var TOP = 18;
  var BOT = 138;
  var MID = (TOP + BOT) / 2;
  var xs = [30, 74, 118, 162];

  function yOf(v) {
    return MID - v * ((BOT - TOP) / 16);
  }

  var s = '<svg width="' + W + '" height="' + H + '" viewBox="0 0 ' + W + ' ' + H + '" xmlns="http://www.w3.org/2000/svg">';

  [-8, -4, 0, 4, 8].forEach(function(g) {
    var y = yOf(g);
    s += '<line x1="16" y1="' + y + '" x2="' + (W - 8) + '" y2="' + y + '" stroke="' + (g === 0 ? '#94a3b8' : '#e2e8f0') + '" stroke-width="' + (g === 0 ? 1.1 : 0.7) + '"/>';
    s += '<text x="5" y="' + (y + 3) + '" font-size="7" fill="#94a3b8">' + g + '</text>';
  });

  var pts = values.map(function(v, i) {
    return xs[i] + ',' + yOf(v);
  }).join(' ');

  s += '<polyline points="' + pts + '" fill="none" stroke="' + accent + '" stroke-width="2" stroke-linejoin="round"/>';

  values.forEach(function(v, i) {
    var y = yOf(v);
    s += '<circle cx="' + xs[i] + '" cy="' + y + '" r="4.2" fill="' + accent + '" stroke="#ffffff" stroke-width="1.2"/>';
    s += '<text x="' + xs[i] + '" y="' + (y - 7) + '" font-size="8" font-weight="700" text-anchor="middle" fill="#334155">' + (Math.round(v * 10) / 10) + '</text>';
    s += '<text x="' + xs[i] + '" y="' + (H - 22) + '" font-size="10" font-weight="800" text-anchor="middle" fill="#0f172a">' + LETTERS[i] + '</text>';
    s += '<text x="' + xs[i] + '" y="' + (H - 10) + '" font-size="6" text-anchor="middle" fill="#64748b">' + ['DOMINANCE', 'INFLUENCE', 'STEADINESS', 'COMPLIANCE'][i] + '</text>';
  });

  s += '</svg>';
  return s;
}

function traitsChips_(traits) {
  if (!traits || !traits.length) return '<span class="muted">-</span>';
  return traits.map(function(t) {
    return '<span class="chip">' + esc_(t) + '</span>';
  }).join(' ');
}

function bulletList_(arr) {
  return arr.map(function(x) {
    return '<li>' + esc_(String(x).replace(/^[•\s]+/, '')) + '</li>';
  }).join('');
}
